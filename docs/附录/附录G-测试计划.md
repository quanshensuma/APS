### G1.2 日期解析（`test_cleaner.py::test_parse_date`）

```python
from datetime import date
from core.cleaner import parse_date, parse_due

@pytest.mark.parametrize("raw,expect,level", [
    ("2026/11/3",  date(2026,11,3),  "exact"),   # 斜杠
    ("2026-11-03", date(2026,11,3),  "exact"),   # 横线
    ("2026.11.4",  date(2026,11,4),  "exact"),   # 点
    ("11月12日",   date(2026,11,12), "exact"),   # 中文（缺年，补 2026）
])
def test_parse_date(raw, expect, level):
    d, lv, note = parse_date(raw)
    assert d == expect
    assert lv == level

@pytest.mark.parametrize("raw,expect_date,expect_prio", [
    ("尽快",   None,             5),      # 无交期但最高优先级
    ("月底前", date(2026,11,30), None),   # 推断为当月末
    ("",       None,             None),   # 空白
])
def test_parse_due_human(raw, expect_date, expect_prio):
    d, level, note, prio = parse_due(raw, ref_year=2026)
    assert d == expect_date
    assert prio == expect_prio
```

**重点**：`11月12日` 缺年份，必须补 `2026`。这是任务书表 4 点名的翻车 #2「交期「11月12日」静默丢失」。

### G1.3 校验器抓撞机（`test_checker.py::test_catches_collision`）

```python
from core.checker import verify_all

def test_catches_collision(clean_mini, opts):
    """同机台两个工序时间重叠 → 必须报 H4。"""
    sched = [
        {"order_id":"MO-001","lot_id":"MO-001","seq":1,"op":"车","group":"车",
         "machine":"LAT-01","qty":10,"duration":200,
         "start_axis":0,"end_axis":200,"status":"scheduled"},
        {"order_id":"MO-002","lot_id":"MO-002","seq":1,"op":"车","group":"车",
         "machine":"LAT-01","qty":10,"duration":200,
         "start_axis":100,"end_axis":300,"status":"scheduled"},  # ← 与上面重叠 100min
    ]
    report = verify_all(sched, clean_mini, opts)
    assert not report.ok()
    assert any(v.rule == "H4" for v in report.h4.violations)
    msg = report.h4.violations[0].message
    assert "LAT-01" in msg
    assert "同一台机两个活同时干" in msg
```

```markdown
| 1 | 不建模换型/调机时间 | 实际完工会比排程晚 10~30% |
| 2 | 不建模人员资质 | 假定人人能干所有机台 |
| 3 | 工时按标准值 | 不考虑效率波动 |
| 4 | 磨工序在台账里点名 GRD-01 | GRD-02 默认用不上，磨缺口 6,590 min |
| 5 | 只排本周（6 天） | 6 张交期在窗口后的单只提示顺延 |
| 6 | 插单只做预演 | 不自动重排 |
```

## G12. 测试盲区（诚实记录）
